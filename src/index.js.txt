const TYPE_UDP = 2;

const ADDR_V6 = 4;

const TEXT_DECODER = new TextDecoder();
let I0, I1, I2, I3, I4, I5, I6, I7, I8, I9, I10, I11, I12, I13, I14, I15;
let CACHED_UUID = '';

const matchID = c =>
    c[1] === I0 && c[2] === I1 && c[3] === I2 && c[4] === I3 &&
    c[5] === I4 && c[6] === I5 && c[7] === I6 && c[8] === I7 &&
    c[9] === I8 && c[10] === I9 && c[11] === I10 && c[12] === I11 &&
    c[13] === I12 && c[14] === I13 && c[15] === I14 && c[16] === I15;

function updateMatchID(uuid) {
    const cleanHex = uuid.replace(/-/g, '');
    const idB = new Uint8Array(16);
    for (let i = 0; i < 16; i++) {
        idB[i] = parseInt(cleanHex.substring(i * 2, i * 2 + 2), 16);
    }
    [I0, I1, I2, I3, I4, I5, I6, I7, I8, I9, I10, I11, I12, I13, I14, I15] = idB;
}

updateMatchID(DEFAULT_UUID);

function isCFError(err) {
    const msg = err?.message?.toLowerCase() || '';
    return msg.includes('proxy request') ||
        msg.includes('cannot connect') ||
        msg.includes('cloudflare');
}


function mergeUint8Array(a, b) {
    const res = new Uint8Array(a.length + b.length);
    res.set(a);
    res.set(b, a.length);
    return res;
}



async function resolveHeader(streamReader) {
    let buffer = new Uint8Array(4096);
    let offset = 0;
    const requireBytes = async (length) => {
        while (offset < length) {
            const {
                value,
                done
            } = await streamReader.read();
            if (done) return false;
            if (offset + value.length > buffer.length) {
                const newSize = Math.max(buffer.length * 2, offset + value.length);
                const newBuffer = new Uint8Array(newSize);
                newBuffer.set(buffer);
                buffer = newBuffer;
            }
            buffer.set(value, offset);
            offset += value.length;
        }
        return true;
    };


    if (!(await requireBytes(18))) return null;
    if (buffer[0] !== PROTO_VER) return {
        error: 'Err:V'
    };
    if (!matchID(buffer)) return {
        error: 'Err:A'
    };

    const metaLen = buffer[17];

    if (!(await requireBytes(22 + metaLen))) return null;

    const action = buffer[18];
    if (action !== TYPE_TCP && action !== TYPE_UDP) return {
        error: `Err:C-${action}`
    };

    const portIdx = 18 + metaLen + 1;
    const port = (buffer[portIdx] << 8) | buffer[portIdx + 1];
    let addrType = buffer[portIdx + 2];
    if (addrType !== 1) addrType += 1;
    const addrStart = portIdx + 3;

    let targetAddr = '';
    let payloadStart = 0;

    if (addrType === ADDR_V4) {
        payloadStart = addrStart + 4;
        if (!(await requireBytes(payloadStart))) return null;
        targetAddr = `${buffer[addrStart]}.${buffer[addrStart+1]}.${buffer[addrStart+2]}.${buffer[addrStart+3]}`;

    } else if (addrType === ADDR_DNS) {
        if (!(await requireBytes(addrStart + 1))) return null;
        const dnsLen = buffer[addrStart];
        payloadStart = addrStart + 1 + dnsLen;
        if (!(await requireBytes(payloadStart))) return null;
        targetAddr = TEXT_DECODER.decode(buffer.subarray(addrStart + 1, payloadStart));

    } else if (addrType === ADDR_V6) {
        payloadStart = addrStart + 16;
        if (!(await requireBytes(payloadStart))) return null;
        const parts = [];
        for (let i = 0; i < 16; i += 2) {
            parts.push(((buffer[addrStart + i] << 8) | buffer[addrStart + i + 1]).toString(16));
        }
        targetAddr = `[${parts.join(':')}]`;
    } else {
        return {
            error: `Err:T-${addrType}`
        };
    }

    const initialData = offset > payloadStart ? buffer.slice(payloadStart, offset) : new Uint8Array(0);
    return {
        targetAddr,
        port,
        initialData,
        ver: buffer[0],
        action
    };
}

// ============================================================================
export default {
    async fetch(request, env, ctx) {
        const _KEY = env.UUID || DEFAULT_UUID;

        // 动态缓存 UUID 转换
        if (_KEY !== CACHED_UUID) {
            CACHED_UUID = _KEY;
            updateMatchID(_KEY);
        }

        // 🚨 白天模式：
        if (request.method === 'GET') {
            // 伪造 Nginx 网页与 Server 响应头
            return new Response(NGINX_HTML, {
                status: 200,
                headers: {
                    'Content-Type': 'text/html; charset=utf-8',
                    'Server': 'nginx'
                }
            });
        }

        // 🌃 黑夜模式：处理真实的代理请求 (POST 流量)
        if (request.method !== 'POST') return new Response('Forbidden', {
            status: 403
        });

        const ingressReader = request.body.getReader();

        try {
            const headerInfo = await resolveHeader(ingressReader);

            if (!headerInfo) {
                try {
                    ingressReader.releaseLock();
                } catch {}
                return new Response('Short Frame', {
                    status: 400
                });
            }
            if (headerInfo.error) {
                try {
                    ingressReader.releaseLock();
                } catch {}
                return new Response(headerInfo.error, {
                    status: 400
                });
            }

            // ==========================================================
            // 🟡 分支 A：UDP 流量 
            // ==========================================================
            if (headerInfo.action === TYPE_UDP) {

                try {
                    ingressReader.releaseLock();
                } catch {}
                return new Response('UDP not supported', {
                    status: 403
                });
            }

            // ==========================================================
            // 🟢 分支 B：TCP 流量 (XHTTP 极致双向中继)
            // ==========================================================
            let egressSocket = null;
            let connectionSuccess = false;
            const fallbackIP = env.PROXY || CF_FALLBACK_IPS[0];

            const targets = [
                headerInfo.targetAddr,
                fallbackIP
            ];


            function connectTimeout(ms) {
                return new Promise((_, reject) => {
                    setTimeout(() => {
                        reject(new Error("Connection timeout"));
                    }, ms);
                });
            }

            for (let i = 0; i < targets.length; i++) {

                try {

                    egressSocket = connect({
                        hostname: targets[i],
                        port: headerInfo.port
                    });

                    await Promise.race([
                        egressSocket.opened,
                        connectTimeout(5000)
                    ]);

                    connectionSuccess = true;
                    break;

                } catch (err) {

                    try {
                        egressSocket?.close();
                    } catch {}

                    const isTimeout =
                        err?.message === 'Connection timeout';

                    if (
                        (isTimeout || isCFError(err)) &&
                        i < targets.length - 1
                    ) {
                        continue;
                    }

                    try {
                        ingressReader.releaseLock();
                    } catch {}

                    return new Response(
                        'Connection Failed', {
                            status: 502
                        }
                    );
                }
            }

            if (!connectionSuccess) {
                try {
                    ingressReader.releaseLock();
                } catch {}
                return new Response('Failed', {
                    status: 502
                });
            }


            const feedbackHead = new Uint8Array([
                headerInfo.ver,
                0
            ]);

            const {
                readable: outputStream,
                writable: internalLink
            } = new TransformStream();



            async function relayUpstream(
                req,
                sock,
                init
            ) {
                if (init.length) {
                    const w = sock.writable.getWriter();
                    try {
                        await w.write(init);
                    } finally {
                        w.releaseLock();
                    }
                }

                return req.body.pipeTo(
                    sock.writable, {
                        preventAbort: true,
                        preventClose: true
                    }
                );
            }

            async function relayDownstream(
                sock,
                link
            ) {
                const w = link.getWriter();
                try {
                    await w.write(
                        feedbackHead
                    );
                } finally {
                    w.releaseLock();
                }

                return sock.readable.pipeTo(
                    link,

                    {
                        preventAbort: true
                    }
                );

            }

            try {
                ingressReader.releaseLock();
            } catch {}

            relayUpstream(
                request,
                egressSocket,
                headerInfo.initialData
            ).catch(() => {

                try {
                    egressSocket.close();
                } catch {}

            });


            relayDownstream(
                egressSocket,
                internalLink
            ).catch(() => {

                try {
                    egressSocket.close();
                } catch {}

            });


            return new Response(
                outputStream, {
                    status: 200,
                    headers: {
                        "Content-Type": "application/octet-stream"
                    }
                }
            );


        } catch (err) {
            try {
                ingressReader.releaseLock();
            } catch {}
            return new Response('Internal Pipeline Error', {
                status: 500
            });
        }
    }
};
