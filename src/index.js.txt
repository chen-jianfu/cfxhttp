    connect
} from 'cloudflare:sockets';

const DEFAULT_UUID = '96c50e3a-5b87-49dd-bd20-03c7f2735e40';
const CF_FALLBACK_IPS = ['ProxyIP.US.CMLiussss.net'];

const PROTO_VER = 0;
const TYPE_TCP = 1;
const TYPE_UDP = 2;
const ADDR_V4 = 1;
const ADDR_DNS = 3;
const ADDR_V6 = 4;

const NGINX_HTML = `<!DOCTYPE html>

const TEXT_DECODER = new TextDecoder();
const CONNECT_TIMEOUT_MS = 3000;
const MAX_CONCURRENT = 64;
let ACTIVE_CONNECTIONS = 0;
let I0, I1, I2, I3, I4, I5, I6, I7, I8, I9, I10, I11, I12, I13, I14, I15;
let CACHED_UUID = '';

const matchID = c =>
    c[1] === I0 && c[2] === I1 && c[3] === I2 && c[4] === I3 &&
    c[5] === I4 && c[6] === I5 && c[7] === I6 && c[8] === I7 &&
    c[9] === I8 && c[10] === I9 && c[11] === I10 && c[12] === I11 &&
    c[13] === I12 && c[14] === I13 && c[15] === I14 && c[16] === I15;

function updateMatchID(uuid) {
    const cleanHex = uuid.replace(/-/g, '');
    const idB = new Uint8Array(16);
    for (let i = 0; i < 16; i++) {
        idB[i] = parseInt(cleanHex.substring(i * 2, i * 2 + 2), 16);
    }
    [I0, I1, I2, I3, I4, I5, I6, I7, I8, I9, I10, I11, I12, I13, I14, I15] = idB;
}

updateMatchID(DEFAULT_UUID);

function isCFError(err) {
    const msg = err?.message?.toLowerCase() || '';
    return msg.includes('proxy request') ||
        msg.includes('cannot connect') ||
        msg.includes('cloudflare');
}


async function readAtLeast(reader, minLen, rlen, cache) {
    if (rlen >= minLen) return { cache, rlen, done: false };
    const r = await reader.readAtLeast(minLen - rlen, new Uint8Array(65536));
    cache.set(r.value, rlen);
    return { cache, rlen: rlen + r.value.length, done: r.done };
}

async function resolveHeader(reader) {
    let r = await reader.readAtLeast(18, new Uint8Array(65536));
    let cache = r.value;
    let rlen = cache.length;
    let done = r.done;

    if (cache[0] !== PROTO_VER) return { error: 'Err:V' };
    if (!matchID(cache)) return { error: 'Err:A' };

    const metaLen = cache[17];
    const needBase = 22 + metaLen;

    if (rlen < needBase) {
        if (done) return null;
        ({ cache, rlen, done } = await readAtLeast(reader, needBase, rlen, cache));
        if (rlen < needBase) return null;
    }

    const action = cache[18 + metaLen];
    if (action !== TYPE_TCP && action !== TYPE_UDP) return { error: `Err:C-${action}` };

    const portIdx = 18 + metaLen + 1;
    const port = (cache[portIdx] << 8) | cache[portIdx + 1];
    let addrType = cache[portIdx + 2];
    if (addrType !== 1) addrType += 1;
    const addrStart = portIdx + 3;

    let targetAddr = '';
    let payloadStart = 0;

    if (addrType === ADDR_V4) {
        payloadStart = addrStart + 4;
        if (rlen < payloadStart) {
            if (done) return null;
            ({ cache, rlen, done } = await readAtLeast(reader, payloadStart, rlen, cache));
            if (rlen < payloadStart) return null;
        }
        targetAddr = `${cache[addrStart]}.${cache[addrStart + 1]}.${cache[addrStart + 2]}.${cache[addrStart + 3]}`;
    } else if (addrType === ADDR_DNS) {
        if (rlen < addrStart + 1) {
            if (done) return null;
            ({ cache, rlen, done } = await readAtLeast(reader, addrStart + 1, rlen, cache));
            if (rlen < addrStart + 1) return null;
        }
        const dnsLen = cache[addrStart];
        payloadStart = addrStart + 1 + dnsLen;
        if (rlen < payloadStart) {
            if (done) return null;
            ({ cache, rlen, done } = await readAtLeast(reader, payloadStart, rlen, cache));
            if (rlen < payloadStart) return null;
        }
        targetAddr = TEXT_DECODER.decode(cache.subarray(addrStart + 1, payloadStart));
    } else if (addrType === ADDR_V6) {
        payloadStart = addrStart + 16;
        if (rlen < payloadStart) {
            if (done) return null;
            ({ cache, rlen, done } = await readAtLeast(reader, payloadStart, rlen, cache));
            if (rlen < payloadStart) return null;
        }
        const parts = [];
        for (let i = 0; i < 16; i += 2) {
            parts.push(((cache[addrStart + i] << 8) | cache[addrStart + i + 1]).toString(16));
        }
        targetAddr = parts.join(':');
    } else {
        return { error: `Err:T-${addrType}` };
    }

    const initialData = rlen > payloadStart ? cache.subarray(payloadStart) : new Uint8Array(0);
    return { targetAddr, port, initialData, ver: cache[0], action };
}

// ============================================================================
export default {
    async fetch(request, env, ctx) {
        const _KEY = env.UUID || DEFAULT_UUID;

        // 动态缓存 UUID 转换
        if (_KEY !== CACHED_UUID) {
            CACHED_UUID = _KEY;
            updateMatchID(_KEY);
        }

        // ?? 白天模式：
        if (request.method === 'GET') {
            // 伪造 Nginx 网页与 Server 响应头
            return new Response(NGINX_HTML, {
                status: 200,
                headers: {
                    'Content-Type': 'text/html; charset=utf-8',
                    'Server': 'nginx'
                }
            });
        }

        // ?? 黑夜模式：处理真实的代理请求 (POST 流量)
        if (request.method !== 'POST') return new Response('Forbidden', {
            status: 403
        });

        if (!request.body) {
            return new Response('Missing request body', { status: 400 });
        }

        if (ACTIVE_CONNECTIONS >= MAX_CONCURRENT) {
            return new Response('Too many connections', { status: 429 });
        }

        ACTIVE_CONNECTIONS++;
        let ingressReader;

        try {
            ingressReader = request.body.getReader({ mode: 'byob' });

            const headerInfo = await resolveHeader(ingressReader);

            if (!headerInfo) {
                try { ingressReader.releaseLock(); } catch {}
                return new Response('Short Frame', { status: 400 });
            }
            if (headerInfo.error) {
                try { ingressReader.releaseLock(); } catch {}
                return new Response(headerInfo.error, { status: 400 });
            }

            // ==========================================================
            // ?? 分支 A：UDP 流量 
            // ==========================================================
            if (headerInfo.action === TYPE_UDP) {
                try { ingressReader.releaseLock(); } catch {}
                return new Response('UDP not supported', { status: 403 });
            }

            // ==========================================================
            // ?? 分支 B：TCP 流量 
            // ==========================================================
            let egressSocket = null;
            let connectionSuccess = false;
            const fallbackIP = env.PROXY || CF_FALLBACK_IPS[0];
            const targets = env.PROXY
                ? [fallbackIP]
                : [headerInfo.targetAddr, fallbackIP];

            async function withTimeout(promise, ms) {
                let timerId;
                const timeoutPromise = new Promise((_, reject) => {
                    timerId = setTimeout(() => reject(new Error('Connection timeout')), ms);
                });
                try {
                    return await Promise.race([promise, timeoutPromise]);
                } finally {
                    clearTimeout(timerId);
                }
            }

            for (let i = 0; i < targets.length; i++) {
                try {
                    egressSocket = connect({
                        hostname: targets[i],
                        port: headerInfo.port
                    }, { allowHalfOpen: true });
                    await withTimeout(egressSocket.opened, CONNECT_TIMEOUT_MS);
                    connectionSuccess = true;
                    break;
                } catch (err) {
                    try { egressSocket?.close(); } catch {}
                    const isTimeout = err?.message === 'Connection timeout';
                    if ((isTimeout || isCFError(err)) && i < targets.length - 1) {
                        continue;
                    }
                    try { ingressReader.releaseLock(); } catch {}
                    return new Response('Connection Failed', { status: 502 });
                }
            }

            const feedbackHead = new Uint8Array([
                headerInfo.ver,
                0
            ]);

            const {
                readable: outputStream,
                writable: internalLink
            } = new TransformStream();

            async function relayUpstream(req, sock, init) {
                if (init.length) {
                    const w = sock.writable.getWriter();
                    try {
                        await w.write(init);
                    } finally {
                        w.releaseLock();
                    }
                }
                return req.body.pipeTo(sock.writable, {
                    preventAbort: false,
                    preventClose: false
                });
            }

            async function relayDownstream(sock, link) {
                const w = link.getWriter();
                try {
                    await w.write(feedbackHead);
                } finally {
                    w.releaseLock();
                }
                return sock.readable.pipeTo(link, { preventAbort: false });
            }

            try { ingressReader.releaseLock(); } catch {}

            const upstreamPromise = relayUpstream(request, egressSocket, headerInfo.initialData);
            const downstreamPromise = relayDownstream(egressSocket, internalLink);

            const cleanupPromise = Promise.allSettled([
                upstreamPromise.catch((err) => {
                    try { egressSocket.close(); } catch {}
                }),
                downstreamPromise.catch((err) => {
                    try { egressSocket.close(); } catch {}
                }),
            ]).finally(() => {
                try { egressSocket.close(); } catch {}
            });

            return new Response(outputStream, {
                status: 200,
                headers: {
                    'Content-Type': 'application/grpc',
                    'X-Accel-Buffering': 'no',
                    'Cache-Control': 'no-store',
                    Connection: 'keep-alive'
                }
            });

        } catch (err) {
            try { ingressReader?.releaseLock(); } catch {}
            return new Response('Internal Pipeline Error', {
                status: 500
            });
        } finally {
            ACTIVE_CONNECTIONS = Math.max(0, ACTIVE_CONNECTIONS - 1);
        }
    }
};
