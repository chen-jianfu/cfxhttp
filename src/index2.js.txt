import { connect } from 'cloudflare:sockets'

// configurations
const UUID = '96c50e3a-5b87-49dd-bd20-03c7f2735e40' // auth token
const FORWARD = 'ProxyIP.US.CMLiussss.net' // (optional) relay for CF websites. e.g. example.com
const LOG_LEVEL = 'info' // debug, info, error, none

// source code
const TIME_ZONE = 8 * 60 * 60 * 1000 // logging timestamp forwards 8 hours
const BUFFER_SIZE = 512 * 1024 // download/upload buffer size in bytes

function to_size(size) {
    const KiB = 1024
    const min = 1.1 * KiB
    let i = 0
    const SIZE_UNITS = ['B', 'KiB', 'MiB', 'GiB', 'TiB']
    for (; i < SIZE_UNITS.length; i++) {
        if (size < min) {
            break
        }
        size = size / KiB
    }
    return `${Math.floor(size)} ${SIZE_UNITS[i]}`
}

function check_token(id, token) {
    for (let index = 0; index < 16; index++) {
        const v = id[index]
        const t = token[index]
        if (v !== t) {
            return false
        }
    }
    return true
}

class Counter {
    #total

    constructor() {
        this.#total = 0
    }

    get() {
        return this.#total
    }

    add(size) {
        this.#total += size
    }
}

function concat_typed_arrays(first, ...args) {
    let len = first.length
    for (let a of args) {
        len += a.length
    }
    const r = new first.constructor(len)
    r.set(first, 0)
    len = first.length
    for (let a of args) {
        r.set(a, len)
        len += a.length
    }
    return r
}

class Logger {
    #id
    #level

    constructor(log_level) {
        this.#id = random_id()

        if (typeof log_level !== 'string') {
            log_level = 'info'
        }
        const levels = ['debug', 'info', 'error', 'none']
        this.#level = levels.indexOf(log_level.toLowerCase())
    }

    debug(...args) {
        if (this.#level < 1) {
            this.#log(`[debug]`, ...args)
        }
    }

    info(...args) {
        if (this.#level < 2) {
            this.#log(`[info ]`, ...args)
        }
    }

    error(...args) {
        if (this.#level < 3) {
            this.#log(`[error]`, ...args)
        }
    }

    #log(prefix, ...args) {
        const now = new Date(Date.now() + TIME_ZONE).toISOString()
        console.log(now, prefix, `(${this.#id})`, ...args)
    }
}

function random_id() {
    const min = 10000
    const max = min * 10 - 1
    return Math.floor(Math.random() * (max - min + 1)) + min
}

function parse_token(token) {
    token = token.replaceAll('-', '')
    const r = []
    for (let index = 0; index < 16; index++) {
        const v = parseInt(token.substr(index * 2, 2), 16)
        r.push(v)
    }
    return r
}

function get_buffer(size) {
    return new Uint8Array(new ArrayBuffer(size || BUFFER_SIZE))
}

// enums
const ADDRESS_TYPE_IPV4 = 1
const ADDRESS_TYPE_URL = 2
const ADDRESS_TYPE_IPV6 = 3

async function read_header(readable, token_str) {
    const reader = readable.getReader({ mode: 'byob' })
    const fail = (msg) => {
        reader.releaseLock()
        return msg
    }

    let r = await reader.readAtLeast(1 + 16 + 1, get_buffer())
    let rlen = 0
    let idx = 0
    let cache = r.value
    rlen += r.value.length

    const version = cache[0]
    const id = cache.slice(1, 1 + 16)
    const token = parse_token(token_str)
    if (!check_token(id, token)) {
        return fail(`invalid UUID`)
    }
    const pb_len = cache[1 + 16]
    const addr_plus1 = 1 + 16 + 1 + pb_len + 1 + 2 + 1

    if (addr_plus1 + 1 > rlen) {
        if (r.done) {
            return fail(`header too short`)
        }
        idx = addr_plus1 + 1 - rlen
        r = await reader.readAtLeast(idx, get_buffer())
        rlen += r.value.length
        cache = concat_typed_arrays(cache, r.value)
    }

    const cmd = cache[1 + 16 + 1 + pb_len]
    if (cmd !== 1) {
        return fail(`unsupported command: ${cmd}`)
    }
    const port = (cache[addr_plus1 - 1 - 2] << 8) + cache[addr_plus1 - 1 - 1]
    const atype = cache[addr_plus1 - 1]
    let header_len = -1
    if (atype === ADDRESS_TYPE_IPV4) {
        header_len = addr_plus1 + 4
    } else if (atype === ADDRESS_TYPE_IPV6) {
        header_len = addr_plus1 + 16
    } else if (atype === ADDRESS_TYPE_URL) {
        header_len = addr_plus1 + 1 + cache[addr_plus1]
    }

    if (header_len < 0) {
        return fail('read address type failed')
    }

    idx = header_len - rlen
    if (idx > 0) {
        if (r.done) {
            return fail(`read address failed`)
        }
        r = await reader.readAtLeast(idx, get_buffer())
        rlen += r.value.length
        cache = concat_typed_arrays(cache, r.value)
    }

    let hostname = ''
    idx = addr_plus1
    switch (atype) {
        case ADDRESS_TYPE_IPV4:
            hostname = cache.slice(idx, idx + 4).join('.')
            break
        case ADDRESS_TYPE_URL:
            hostname = new TextDecoder().decode(
                cache.slice(idx + 1, idx + 1 + cache[idx]),
            )
            break
        case ADDRESS_TYPE_IPV6:
            hostname = cache
                .slice(idx, idx + 16)
                .reduce(
                    (s, b2, i2, a) =>
                        i2 % 2
                            ? s.concat(((a[i2 - 1] << 8) + b2).toString(16))
                            : s,
                    [],
                )
                .join(':')
            break
    }

    if (hostname.length < 1) {
        return fail('failed to parse hostname')
    }

    const data = cache.slice(header_len)
    return {
        hostname,
        port,
        data,
        resp: new Uint8Array([version, 0]),
        reader,
        done: r.done,
    }
}

async function forward_upstream(counter, log, writer, session) {
    async function inner_upload(d, src) {
        if (!d || d.length < 1) {
            log.debug(`upload skip empty ${src}`)
            return
        }
        counter.add(d.length)
        log.debug(`upload ${src}: ${to_size(d.length)}`)
        await writer.write(d)
    }

    await inner_upload(session.data, 'first packet')
    while (!session.done) {
        const r = await session.reader.read(get_buffer())
        if (r.value) {
            await inner_upload(r.value, 'remain packets')
        }
        session.done = r.done
    }
}

function create_sender(log, session, writable) {
    const counter = new Counter()
    const done = new Promise((resolve, reject) => {
        const writer = writable.getWriter()
        forward_upstream(counter, log, writer, session)
            .then(resolve)
            .catch(reject)
            .finally(() => {
                writer
                    .close()
                    .then(() => log.debug(`upload writer closed`))
                    .catch((err) => log.debug(`upload writer error: ${err}`))
            })
    })

    return {
        counter,
        done,
    }
}

function create_receiver(log, resp, remote_readable) {
    const counter = new Counter()
    let stream

    const done = new Promise((resolve, reject) => {
        stream = new TransformStream(
            {
                start(controller) {
                    log.debug(`copy protocol response`)
                    counter.add(resp.length)
                    controller.enqueue(resp)
                },
                transform(chunk, controller) {
                    counter.add(chunk.length)
                    controller.enqueue(chunk)
                    log.debug(`receive: ${to_size(chunk.length)}`)
                },
                cancel(reason) {
                    reject(`download cancelled: ${reason}`)
                },
            },
            null,
            new ByteLengthQueuingStrategy({ highWaterMark: BUFFER_SIZE }),
        )
        remote_readable.pipeTo(stream.writable).catch(reject).finally(resolve)
    })

    return {
        readable: stream.readable,
        counter,
        done,
    }
}

async function open_channel(log, session, ...remotes) {
    const hostname = remotes.shift()
    if (!hostname || hostname.length < 1) {
        log.info('all attempts failed')
        return null
    }

    if (session.hostname === hostname) {
        log.info(`direct connect [${session.hostname}]:${session.port}`)
    } else {
        log.info(`tunnel [${session.hostname}]:${session.port} via ${hostname}`)
    }

    const retry = () => open_channel(log, session, ...remotes)
    let remote
    try {
        remote = connect({ hostname: hostname, port: session.port })
        const info = await remote.opened
        log.debug(`connection opened:`, info)
    } catch (err) {
        log.error(`retry [${session.hostname}] reason: ${err}`)
        return await retry()
    }

    const sender = create_sender(log, session, remote.writable)
    const receiver = create_receiver(log, session.resp, remote.readable)
    return {
        receiver,
        sender,
    }
}

async function handle_tunnel(log, body, cfg) {
    const session = await read_header(body, cfg.UUID)
    if (typeof session !== 'object' || !session) {
        log.error(`failed to parse header: ${session}`)
        return null
    }

    const r = await open_channel(log, session, session.hostname, cfg.FORWARD)
    if (r === null) {
        log.error('create remote stream failed')
        return null
    }

    const connection_closed = new Promise((resolve, _) => {
        r.receiver.done
            .then(() => log.debug(`download complete`))
            .catch((err) => log.error(`download error: ${err}`))
            .finally(() => r.sender.done)
            .then(() => log.debug(`upload complete`))
            .catch((err) => log.debug(`upload error: ${err}`))
            .finally(() => {
                const total_upload = to_size(r.sender.counter.get())
                const total_download = to_size(r.receiver.counter.get())
                log.info(
                    `connection closed. uploaded: ${total_upload} downloaded: ${total_download}`,
                )
                resolve()
            })
    })

    return {
        readable: r.receiver.readable,
        closed: connection_closed,
    }
}

async function handle_action(request, cfg) {
    const log = new Logger(cfg.LOG_LEVEL)
    try {
        return await handle_tunnel(log, request.body, cfg)
    } catch (err) {
        log.error(`error: ${err}`)
    }
    return null
}

function create_config(url, uuid) {
    const config = JSON.parse(config_template)
    const out = config['outbounds'][0]['settings']['vnext'][0]
    const stream = config['outbounds'][0]['streamSettings']

    // workers are TLS only!
    const host = url.hostname
    const path = url.pathname
    out['address'] = host
    out['users'][0]['id'] = uuid
    stream['xhttpSettings']['host'] = host
    stream['xhttpSettings']['path'] = path.endsWith('/') ? path : `${path}/`
    stream['tlsSettings']['serverName'] = host

    return JSON.stringify(config)
}

async function fetch(request, env, ctx) {
    const cfg = {
        UUID: env.UUID || UUID,
        FORWARD: env.FORWARD || FORWARD,
        LOG_LEVEL: env.LOG_LEVEL || LOG_LEVEL,
    }

    if (!cfg.UUID) {
        return new Response(`Error: UUID is empty`)
    }

    if (request.method === 'POST') {
        const r = await handle_action(request, cfg)
        if (r) {
            ctx.waitUntil(r.closed)
            return new Response(r.readable, {
                headers: {
                    'X-Accel-Buffering': 'no',
                    'Cache-Control': 'no-store',
                    Connection: 'Keep-Alive',
                    'User-Agent': 'Go-http-client/2.0',
                    'Content-Type': 'application/grpc',
                    // 'Content-Type': 'text/event-stream',
                    // 'Transfer-Encoding': 'chunked',
                },
            })
        }
        return new Response('Bad Request', { status: 400 })
    }

    if (request.method === 'GET') {
        const url = new URL(request.url)
        const items = [url.pathname, url.search]
        for (let item of items) {
            if (item.indexOf(`${cfg.UUID}`) >= 0) {
                const config = create_config(url, cfg.UUID)
                return new Response(config, {
                    headers: {
                        'Content-Type': 'application/json',
                    },
                })
            }
        }
    }
    return new Response(`Hello world!`)
}
